    OUT_PATIENT
  }

  class InsuranceDetails {
    - policyNumber: String
    - providerName: String
    - groupNumber: String
    - memberId: String
    + getPolicyNumber(): String
    + getProviderName(): String
  }

  class GovernmentProgramEnrollment {

    + getEnrollmentId(): String

  Patient "1" -- "0..1" InsuranceDetails : has >

  class ProcedureCharge extends Charge {

    + getProcedureCode(): String
  }

  class MedicationCharge extends Charge {
    - medicationName: String
    - dosage: String
    + getMedicationName(): String
  }

  class DiagnosticTestCharge extends Charge {
    - testCode: String
    - facility: String
    + getTestCode(): String
  }

  class RoomCharge extends Charge {
    - roomNumber: String
    - daysOccupied: int
    + getRoomNumber(): String
    + getDaysOccupied(): int
  }

  class Money {
    - amount: BigDecimal
    - currency: String ' e.g., "USD"

  Charge *-- "1" Money : has baseAmount >

package "Billing Policies and Rules" {
  interface BillingPolicy {
    + calculatePatientPortion(bill: Bill, charges: List<Charge>, patient: Patient): Money
  }

  class InsuranceBillingPolicy implements BillingPolicy {
    - insuranceCoverageRule: InsuranceCoverageRule
    + calculatePatientPortion(bill: Bill, charges: List<Charge>, patient: Patient): Money
  }

  class GovernmentBillingPolicy implements BillingPolicy {
    - subsidizedRateRule: SubsidizedRateRule
    + calculatePatientPortion(bill: Bill, charges: List<Charge>, patient: Patient): Money
  }

  class SelfPayBillingPolicy implements BillingPolicy {
    + calculatePatientPortion(bill: Bill, charges: List<Charge>, patient: Patient): Money
  }

  class InsuranceCoverageRule {
    - ruleId: String
    - coverageDetails: String
    - reimbursementLimit: Money
    - coPaymentStructure: String
    + applyCoverage(charge: Charge, patient: Patient): Money
    + getReimbursementLimit(): Money
  }

  class SubsidizedRateRule {
    - ruleId: String
    - programName: String
    - fixedRate: Money
    + applySubsidizedRate(charge: Charge, patient: Patient): Money
    + getFixedRate(): Money
  }

  InsuranceBillingPolicy "1" *-- "1" InsuranceCoverageRule : applies >
  GovernmentBillingPolicy "1" *-- "1" SubsidizedRateRule : applies >

  enum BillingPolicyType {
    INSURANCE
    GOVERNMENT
    SELF_PAY
  }

  note right of BillingPolicy : Strategy Pattern: policies are interchangeable at runtime.
}
