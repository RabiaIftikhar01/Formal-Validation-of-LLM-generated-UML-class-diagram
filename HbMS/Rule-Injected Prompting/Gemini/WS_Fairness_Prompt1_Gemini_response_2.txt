    + calculatePatientObligation(bill: Bill, patient: Patient): void

  note top of IBillingPolicy : Policies are interchangeable at runtime (Strategy Pattern)\nfor different billing rules.

  class SelfPayBillingPolicy implements IBillingPolicy {
    + calculatePatientObligation(bill: Bill, patient: Patient): void
  }

  class InsuranceBillingPolicy implements IBillingPolicy {
    + calculatePatientObligation(bill: Bill, patient: Patient): void
  }

  class GovernmentBillingPolicy implements IBillingPolicy {
    + calculatePatientObligation(bill: Bill, patient: Patient): void
  }

  class InsuranceCoverage {

    + providerName: String
    + reimbursementLimit: Money
    + coPaymentRate: Decimal

    + getCoPaymentRate(): Decimal

    --
    + getProgramName(): String
  }

    + calculatePrice(basePrice: Money): Money

    + calculatePrice(basePrice: Money): Money
  }

  class DiscountPricingRule implements IPricingRule {
    + calculatePrice(basePrice: Money): Money
