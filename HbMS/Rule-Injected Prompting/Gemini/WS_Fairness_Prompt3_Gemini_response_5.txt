' ----------------------------------------------------------------------------------------------------
' SECTION: Billing Policies (Strategy Pattern for dynamic interchangeability)
' ----------------------------------------------------------------------------------------------------
abstract class BillingPolicy {
    + calculatePatientPortion(totalAmount: Money, insurance: InsuranceDetails): Money
}

class InsuredPatientPolicy {
    + calculatePatientPortion(totalAmount: Money, insurance: InsuranceDetails): Money
}

class GovernmentProgramPolicy {
    + subsidyRate: Percentage
    + calculatePatientPortion(totalAmount: Money, insurance: InsuranceDetails): Money
}

class SelfFundedPolicy {
    + calculatePatientPortion(totalAmount: Money, insurance: InsuranceDetails): Money
}

' Relationships for Billing Policies
BillingPolicy <|-- InsuredPatientPolicy
BillingPolicy <|-- GovernmentProgramPolicy
BillingPolicy <|-- SelfFundedPolicy

note "Strategy Pattern: Allows dynamic interchangeability of billing logic at runtime.\nNew policies can be added without modifying existing code." as N_BillingPolicyStrategy
BillingPolicy .. N_BillingPolicyStrategy

' ----------------------------------------------------------------------------------------------------
' SECTION: Treatment and Pricing (Extensible cost sources and pricing rules)
' ----------------------------------------------------------------------------------------------------

    + getTreatmentId(): String
    + getName(): String
    + getBaseCost(): Money
}

    + applyRule(baseCost: Money, quantity: int): Money
}

    + procedureCode: String

    + getBaseCost(): Money
}

    + drugName: String
    + dosage: String
    + getBaseCost(): Money
}

    + serviceCode: String

    + getBaseCost(): Money
}

class Accommodation implements Treatment {
    + roomNumber: String

    + numberOfDays: int
    + getBaseCost(): Money
}

' Concrete Pricing Rules
class StandardPricingRule extends PricingRule {
    + baseRate: Money
    + applyRule(baseCost: Money, quantity: int): Money
}

class VolumeDiscountRule extends PricingRule {
    + discountThreshold: int
    + discountPercentage: Percentage
    + applyRule(baseCost: Money, quantity: int): Money
}
