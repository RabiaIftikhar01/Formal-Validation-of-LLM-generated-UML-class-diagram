' --- Patient and Bill Data ---
class Patient {
    - patientId : String
    - name : String
    - patientType : PatientType <<enum>>

    -- Operations --
    + getPatientId() : String
    + getName() : String
    + getPatientType() : PatientType
    + getInsuranceCoverage() : InsuranceCoverage
}

    INSURED
    GOVERNMENT_SPONSORED
    SELF_PAY
}

    - billId : String
    - issueDate : Date
    - totalAmount : decimal
    -- Associations --
    - patient : Patient

    -- Operations --
    + getBillId() : String
    + getIssueDate() : Date
    + getTotalAmount() : decimal
    + addEntry(entry : BillingEntry) : void
    + calculateSubtotal() : decimal
    + setTotalAmount(amount : decimal) : void
}

class BillingEntry {
    - entryId : String

    -- Associations --
    - pricingRule : PricingRule
    -- Operations --
    + getEntryId() : String
    + getDescription() : String
    + getQuantity() : int
    + getUnitPrice() : decimal
    + calculateLineItemAmount() : decimal
}

    + applyPricing(baseAmount : decimal, quantity : int) : decimal

    + applyPricing(baseAmount : decimal, quantity : int) : decimal

    + applyPricing(baseAmount : decimal, quantity : int) : decimal

' --- Policy and Program Details ---
