' --- Cost Items (Polymorphic Aggregation) ---
abstract class CostItem {
    - itemId : String
    - description : String
    - baseAmount : Money
    -- Operations --
    + {abstract} calculateAmount() : Money
    + getDescription() : String
    + getBaseAmount() : Money
}

class MedicalProcedure extends CostItem {
    - procedureCode : String
    -- Operations --
    + calculateAmount() : Money
}

class Medication extends CostItem {
    - drugName : String
    - dosage : String
    -- Operations --
    + calculateAmount() : Money
}

class DiagnosticService extends CostItem {
    - serviceType : String
    -- Operations --
    + calculateAmount() : Money
}

class AccommodationCharge extends CostItem {
    - roomType : String
    - numNights : int
    -- Operations --
    + calculateAmount() : Money
}

HospitalStay "1" o-- "*" CostItem : aggregates >
note right of CostItem : Each CostItem encapsulates its own pricing logic via calculateAmount().
