' Design Notes for clarity and adherence to principles
note "Strategy Pattern for flexible and runtime-replaceable billing policies." as N1
note "Abstract 'Charge' and its specializations model various billable items (Composite-like structure)." as N2
note "ChargeEntry acts as an association class, linking an Invoice to a Charge with an adjusted amount specific to that invoice." as N3

interface BillingPolicy {
    + calculateAdjustedAmount(rawAmount : double, patient : Patient) : double
}
N1 .u. BillingPolicy

class InsurancePolicy implements BillingPolicy {
    ' Operations to apply insurance-specific rules
    + calculateAdjustedAmount(rawAmount : double, patient : Patient) : double
}

class GovernmentPolicy implements BillingPolicy {
    ' Operations to apply government-subsidized rates
    + calculateAdjustedAmount(rawAmount : double, patient : Patient) : double
}

class SelfPayPolicy implements BillingPolicy {
    ' Operations to apply full charge amount
    + calculateAdjustedAmount(rawAmount : double, patient : Patient) : double
}

class Patient {
    - patientId : String
    - name : String
    - classification : String ' e.g., "INSURED", "GOVERNMENT", "SELF_PAY"
    + getPatientId() : String
    + getName() : String
    + getClassification() : String
}

class InsuranceCoverage {
    - policyNumber : String
    - insurerName : String
    - coverageDetails : Map<String, Double> ' e.g., serviceType -> coveragePercentage
    - coPayAmount : double
    + isCovered(serviceType : String) : boolean
    + getCoveragePercentage(serviceType : String) : double
    + getCoPayAmount() : double
}
