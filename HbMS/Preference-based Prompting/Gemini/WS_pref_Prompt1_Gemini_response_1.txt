' Interfaces and Abstract Classes for Strategy Pattern and Component Hierarchy

interface BillingPolicy {
    + calculateFinalAmount(bill : Bill, patient : Patient) : double
}

abstract class ChargeItem {
    - description : String
    - baseAmount : double
    + {abstract} getAmount() : double
    + getDescription() : String
}

' Concrete Implementations of Billing Policies (Strategies)

class InsuranceBillingPolicy implements BillingPolicy {
    - {static} DEFAULT_COPAY_RATE : double = 0.2
    - {static} DEFAULT_REIMBURSEMENT_LIMIT : double = 50000.0
    + calculateFinalAmount(bill : Bill, patient : Patient) : double
}

class GovernmentBillingPolicy implements BillingPolicy {
    - {static} SUBSIDY_FACTOR : double = 0.7
    + calculateFinalAmount(bill : Bill, patient : Patient) : double
}

class SelfPayBillingPolicy implements BillingPolicy {
    + calculateFinalAmount(bill : Bill, patient : Patient) : double
}

' Concrete Implementations of Charge Items (Bill Components)

class ProcedureCharge extends ChargeItem {
    - procedureCode : String
    + getAmount() : double
}

class MedicationCharge extends ChargeItem {
    - medicationName : String
    - dosage : String
    - quantity : int
    + getAmount() : double
}

class DiagnosticCharge extends ChargeItem {
    - testName : String
    - testCode : String
    + getAmount() : double
}

class RoomCharge extends ChargeItem {
    - roomNumber : String
    - dailyRate : double
    - daysStayed : int
    + getAmount() : double
}

' Core Domain Classes

class Patient {
    - patientId : String
    - name : String
    - admissionDate : Date
    - dischargeDate : Date
    - billingType : BillingType
    + getBillingType() : BillingType
    + getInsuranceCoverage() : InsuranceCoverage
}

enum BillingType {
    INSURANCE
    GOVERNMENT
    SELF_PAY
}

class InsuranceCoverage {
    - policyNumber : String
    - providerName : String
    - planName : String
    - coverageDetails : String
    - reimbursementLimit : double
    - coPaymentRate : double
    + getReimbursementLimit() : double
    + getCoPaymentRate() : double
    + hasCoverage() : boolean
}

class Bill {
    - billId : String
    - issueDate : Date
    - totalRawAmount : double
    - totalDueAmount : double
    - charges : List<ChargeItem>
    + addCharge(item : ChargeItem)
    + calculateRawTotal() : double
    + setTotalDueAmount(amount : double)
    + getTotalDueAmount() : double
}

class BillProcessor {
    + generateDischargeBill(patient : Patient, chargeItems : List<ChargeItem>) : Bill
    - determineBillingPolicy(patient : Patient) : BillingPolicy
}

' Relationships

' Patient and Insurance
Patient "1" --> "0..1" InsuranceCoverage : has

' Bill composition
Bill "1" *-- "1..*" ChargeItem : includes

' Bill associated with Patient (unidirectional for clarity)
Patient "1" <-- "1" Bill : for

' BillProcessor dependencies and interactions
BillProcessor ..> Bill : creates
BillProcessor ..> Patient : uses
BillProcessor ..> ChargeItem : uses
BillProcessor ..> BillingPolicy : determines and uses

' Billing Policy accessing necessary information for calculation
BillingPolicy ..> Bill : accesses
BillingPolicy ..> Patient : accesses
InsuranceBillingPolicy ..> InsuranceCoverage : accesses (via Patient)

' Explanatory Notes for clarity and understanding
note right of BillProcessor : Orchestrates the discharge bill generation process.\nIt determines the appropriate BillingPolicy\nbased on patient information and applies it.
note right of BillingPolicy : Defines the strategy for calculating\nthe final bill amount. Allows billing rules\nto be interchangeable at runtime.
note top of ChargeItem : Represents a single billable service or item.\nEach concrete ChargeItem encapsulates its\nown pricing logic and details.
note bottom of Patient : Stores core patient information and indicates\nthe type of billing policy applicable.
