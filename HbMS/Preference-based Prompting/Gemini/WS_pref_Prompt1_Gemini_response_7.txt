' Pricing Rules (Strategy Pattern)
interface IPricingRule <<Strategy>> {
    + {abstract} applyPricing(baseAmount : decimal) : decimal
}

class StandardPricingRule {
    + applyPricing(baseAmount : decimal) : decimal
}

class DiscountPricingRule {
    - discountPercentage : decimal
    + applyPricing(baseAmount : decimal) : decimal
}

IPricingRule <|.. StandardPricingRule
IPricingRule <|.. DiscountPricingRule

' Charge Items (Composite pattern can be applied, but for simplicity, keeping as individual items)
abstract class ChargeItem {
    - description : string
    - baseAmount : decimal
    # pricingRule : IPricingRule
    + {abstract} calculateAmount() : decimal
    + getDescription() : string
    + getBaseAmount() : decimal
    + setPricingRule(rule : IPricingRule) : void
}

class ProcedureCharge {
    - procedureCode : string
    + calculateAmount() : decimal
}

class MedicationCharge {
    - medicationName : string
    - dosage : string
    + calculateAmount() : decimal
}

class DiagnosticTestCharge {

    + calculateAmount() : decimal
}

class RoomOccupancyCharge {

    - numberOfDays : int
    + calculateAmount() : decimal

' Billing Policies (Strategy Pattern)
interface IBillingPolicy <<Strategy>> {
    + {abstract} applyPolicy(patient : Patient, chargeItems : List<ChargeItem>) : decimal
}

class InsuranceBillingPolicy {
    + applyPolicy(patient : Patient, chargeItems : List<ChargeItem>) : decimal
}

class GovernmentProgramBillingPolicy {
    + applyPolicy(patient : Patient, chargeItems : List<ChargeItem>) : decimal
}

class SelfPayBillingPolicy {
    + applyPolicy(patient : Patient, chargeItems : List<ChargeItem>) : decimal
}

IBillingPolicy <|.. InsuranceBillingPolicy
IBillingPolicy <|.. GovernmentProgramBillingPolicy
IBillingPolicy <|.. SelfPayBillingPolicy
