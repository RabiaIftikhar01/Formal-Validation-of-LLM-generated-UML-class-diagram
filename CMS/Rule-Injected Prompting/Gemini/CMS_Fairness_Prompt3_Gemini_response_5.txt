hide empty members

' --- Enums ---
enum EnrollmentStatus {
  Waitlisted
  Active
  Completed
}

enum CourseLevelType {
  Undergraduate
  Postgraduate
}

enum NotificationType {
  Announcement
  AssignmentDeadline
  SubmissionReminder
  GeneralUpdate
}

' --- Core Classes ---

' Abstract Base Class for Users
abstract class User {
  -userId: String
  -name: String
  -email: String
  +getUserInfo(): String
  +receiveNotification(notification: Notification): void
}

' Concrete User Roles
class Student {
  +browseAvailableCourses(): List<CourseOffering>
  +enrollInCourse(offering: CourseOffering): Enrollment
  +viewCourseContent(material: CourseMaterial): String
  +trackEnrollmentStatus(enrollment: Enrollment): EnrollmentStatus
}

class Instructor {
  +createCourseOffering(template: CourseTemplate, year: int, semester: String, level: CourseLevelType): CourseOffering
  +updateCourseMaterial(material: CourseMaterial, newContent: String): void
  +publishAnnouncement(offering: CourseOffering, message: String): void
  +sendNotification(notification: Notification, recipients: List<User>): void
}

' Course Definitions and Offerings
class CourseTemplate {
  -templateId: String
  -title: String
  -description: String
  +getCourseDetails(): String
}

class CourseOffering {
  -offeringId: String
  -year: int
  -semester: String
  -level: CourseLevelType
  +getOfferingDetails(): String
  +addCourseMaterial(material: CourseMaterial): void
}
note top of CourseOffering : Represents a specific instance of a course offered in a given semester/year, taught by an instructor.

class CourseMaterial {
  -materialId: String
  -title: String
  -content: String
  -uploadDate: Date
  +viewContent(): String
  +updateContent(newContent: String): void
}

' Enrollment (Association Class)
class Enrollment {
  -enrollmentDate: Date
  -status: EnrollmentStatus
  +updateStatus(newStatus: EnrollmentStatus): void
  +getEnrollmentDetails(): String
}
note top of Enrollment : An association class capturing specific details (date, status) of a student's enrollment in a course offering.

' Notifications
class Notification {
  -notificationId: String
  -message: String
  -timestamp: Date
  -type: NotificationType
  +getNotificationDetails(): String
}

' --- Services ---
class NotificationService {
  +sendNotificationToUser(notification: Notification, recipient: User): void
  +sendNotificationToCourseStudents(notification: Notification, offering: CourseOffering): void
}
