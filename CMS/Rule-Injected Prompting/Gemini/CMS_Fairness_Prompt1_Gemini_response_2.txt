```plantuml
@startuml
!pragma layout smetana

' --- Enums ---
' Naming convention: PascalCase for enum names and values.
enum EnrollmentStatus {
  Waitlisted
  Active
  Completed
}

enum CourseType {
  Undergraduate
  Postgraduate
}

enum NotificationType {
  AssignmentDeadline
  Announcement
  Submission
  General
}

' --- Abstract Classes ---
' Naming convention: PascalCase for class names, camelCase for attributes/methods.
' Attributes are private by default, accessors are public.
abstract class User {
  - userId: String
  - name: String
  - email: String
  -- Operations --
  + getUserId(): String
  + getName(): String
  + getEmail(): String
  + abstract displayProfile(): String
  + receiveNotification(notification: Notification): void
}

abstract class Course {
  - courseId: String
  - title: String
  - description: String
  -- Operations --
  + getCourseId(): String
  + getTitle(): String
  + getDescription(): String
  + viewCourseDetails(): String
}

' --- Concrete Classes ---
class Student {
  -- Operations --
  + browseCourses(): List<CourseOffering>
  + enrollInCourse(courseOffering: CourseOffering): Enrollment
  + viewAnnouncements(): List<Notification>
  + trackEnrollmentStatus(): List<Enrollment>
  + viewCourseContent(material: CourseMaterial): String
}

class Instructor {
  -- Operations --
  + createCourseOffering(course: Course, semester: String, year: int, courseType: CourseType): CourseOffering
  + updateCourseMaterial(courseMaterial: CourseMaterial, newContent: String): void
  + sendNotification(notification: Notification, recipients: List<User>): void
}

class CourseOffering {
  - offeringId: String
  - semester: String
  - year: int
  - maxCapacity: int
  - courseType: CourseType
  -- Operations --
  + getOfferingId(): String
  + getSemester(): String
  + getYear(): int
  + getMaxCapacity(): int
  + getCourseType(): CourseType
  + publish(): void
  + addMaterial(material: CourseMaterial): void
  + removeMaterial(materialId: String): void
  + getEnrolledStudents(): List<Student>
  + getCourseMaterials(): List<CourseMaterial>
}

class CourseMaterial {
  - materialId: String
  - title: String
  - content: String // e.g., URL, file path, or direct text
  - uploadDate: Date
  -- Operations --
  + getMaterialId(): String
  + getTitle(): String
  + getContent(): String
  + getUploadDate(): Date
  + updateContent(newContent: String): void
}

' Enrollment is an Association Class between Student and CourseOffering
class Enrollment {
  - enrollmentDate: Date
  - status: EnrollmentStatus
  -- Operations --
  + getEnrollmentDate(): Date
  + getStatus(): EnrollmentStatus
  + updateStatus(newStatus: EnrollmentStatus): void
}

class Notification {
  - notificationId: String
  - message: String
  - timestamp: Date
  - type: NotificationType
  - sender: User
  - contextCourseOffering: CourseOffering
  -- Operations --
  + getNotificationId(): String
  + getMessage(): String
  + getTimestamp(): Date
  + getType(): NotificationType
  + getSender(): User
  + getContextCourseOffering(): CourseOffering
}

class NotificationService {
  -- Operations --
  + sendNotification(notification: Notification, recipients: List<User>): void
}

' --- Relationships ---

' Inheritance
User <|-- Student
User <|-- Instructor
Course <|-- CourseOffering : is_an_instance_of

' Associations with Multiplicity and Roles
Instructor "1" -- "0..*" CourseOffering : creates >
' An Instructor creates CourseOfferings.

Course "1" -- "0..*" CourseOffering : offered_as >
' A Course can be offered as multiple CourseOfferings.

CourseOffering "1" -- "0..*" CourseMaterial : contains >
' A CourseOffering contains multiple CourseMaterials.

Student "1" -- "0..*" Enrollment : has_enrollments >
' A Student can have multiple Enrollments.

CourseOffering "1" -- "0..*" Enrollment : has_enrollments >
' A CourseOffering can have multiple Enrollments.
' Enrollment class models the details of this association.

Instructor "1" --> "0..*" Notification : sends >
' An Instructor sends Notifications.

Notification "1" --> "1" User : has_sender >
' Each Notification has exactly one sender.

Notification "1" --> "0..1" CourseOffering : relates_to >
' A Notification can optionally relate to a specific CourseOffering.

User "0..*" <-- "1" Notification : receives <
' A Notification can be received by multiple Users.

' Dependencies (uses)
Instructor ..> NotificationService : uses
' Instructor uses NotificationService to send notifications.

Student ..> CourseOffering : views
' Student views CourseOfferings.

Student ..> CourseMaterial : views
' Student views CourseMaterials.

' --- Explanatory Notes ---
note "Course represents the conceptual definition of a subject (e.g., 'Calculus I').\nCourseOffering represents a specific instance of that course offered in a particular semester/year (e.g., 'Calculus I - Fall 2023 - Undergraduate'). This design avoids deep inheritance for course types and promotes flexibility." as N1
Course .. N1
CourseOffering .. N1

note "Enrollment is modeled as an association class to encapsulate attributes specific to the relationship between a Student and a CourseOffering, such as `enrollmentDate` and `status`." as N2
Enrollment .. N2

note "NotificationService is a utility class responsible for the actual mechanism of delivering notifications. Instructors initiate the notification process, but the service handles the technical dispatch." as N3
NotificationService .. N3
Instructor .. N3

note "The `updateCourseMaterial` method in `Instructor` delegates the actual content update to the `CourseMaterial` object, ensuring `CourseMaterial` manages its own data." as N4
Instructor .. N4
CourseMaterial .. N4
```
