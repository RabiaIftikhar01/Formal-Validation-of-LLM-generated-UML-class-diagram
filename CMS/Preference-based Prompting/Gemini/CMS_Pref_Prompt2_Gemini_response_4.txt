    - id: String
    - name: String
    - email: String
    + {abstract} getInfo(): String
    + getId(): String
    + getName(): String
    + getEmail(): String
}

class Student {
    ' Operations that a Student initiates, typically delegated to services
    + browseCourses(): List<Course>
    + enrollInCourse(course: Course): Enrollment
    + viewAnnouncements(): List<Notification>
    + trackEnrollmentStatus(): EnrollmentStatus
    + viewCourseMaterial(course: Course): List<CourseMaterial>
}

    ' Operations that an Instructor initiates, typically delegated to services
    + createCourse(title: String, description: String, category: CourseCategory): Course
    + updateCourseContent(course: Course, material: CourseMaterial): void
    + sendNotification(course: Course, subject: String, message: String): void
}

    + getDescription(): String

    + getInfo(): String
}

class PostgraduateCourse extends Course {
    + getInfo(): String
}

    Undergraduate
    Postgraduate
}

' Relationship: An Instructor publishes/manages multiple Courses

' --- 4. Enrollment (Association Class) ---
