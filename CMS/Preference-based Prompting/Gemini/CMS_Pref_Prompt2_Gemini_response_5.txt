@startuml
' Set a consistent naming convention and style
hide circle
skinparam classAttributeIconSize 0
skinparam classArrowFontSize 10
skinparam defaultFontSize 12
skinparam monochrome true
skinparam shadowing false

' Enums for clarity and type safety
enum EnrollmentStatus {
  Waitlisted
  Active
  Completed
}

enum NotificationType {
  Announcement
  AssignmentDeadline
  Submission
}

' --------------------------
' Core Entities and Roles
' --------------------------

' User represents an individual in the system
class User {
  - id : String
  - name : String
  - email : String
  --
  + getInfo() : String
  + addRole(role: UserRole)
  + removeRole(role: UserRole)
  + hasRole(roleType: String) : boolean
}

' Abstract UserRole to define common role interface
abstract class UserRole {
  + {abstract} getDescription() : String
  + {abstract} canPerform(action: String) : boolean
}

' Specific roles extending UserRole
class StudentRole extends UserRole {
  + getDescription() : String
  + canPerform(action: String) : boolean
  + browseCourses()
  + enrollInCourse(course: Course) : Enrollment
  + viewAnnouncements()
  + trackEnrollmentStatus(enrollment: Enrollment)
}

class InstructorRole extends UserRole {
  + getDescription() : String
  + canPerform(action: String) : boolean
  + publishCourse(course: Course)
  + updateCourseContent(course: Course, resource: CourseResource)
  + sendNotification(notification: Notification, recipients: List<User>)
}

' User has multiple roles (composition)
User "1" *-- "0..*" UserRole : has >

' --------------------------
' Course Management
' --------------------------

' Abstract Course for common course properties
abstract class Course {
  - id : String
  - title : String
  - description : String
  --
  + {abstract} getCourseCategory() : String
  + getInfo() : String
  + addResource(resource: CourseResource)
  + removeResource(resource: CourseResource)
  + getResources() : List<CourseResource>
}

' Specific course categories
class UndergraduateCourse extends Course {
  + getCourseCategory() : String
}

class PostgraduateCourse extends Course {
  + getCourseCategory() : String
}

' CourseResource holds learning materials
class CourseResource {
  - id : String
  - title : String
  - contentPath : String ' e.g., URL or file path
  - uploadedDate : Date
  --
  + viewContent() : String
  + getTitle() : String
}

' Course contains multiple resources (composition)
Course "1" *-- "0..*" CourseResource : contains >

' --------------------------
' Enrollment
' --------------------------

' Enrollment is an association class between User and Course
class Enrollment {
  - enrollmentDate : Date
  - status : EnrollmentStatus
  --
  + updateStatus(newStatus: EnrollmentStatus)
  + getEnrollmentDate() : Date
  + getStatus() : EnrollmentStatus
}

' Relationship between User, Course, and Enrollment
User "1" -- "*" Enrollment : enrolls >
Course "1" -- "*" Enrollment : has >

' --------------------------
' Notification System
' --------------------------

' Notification represents a message to be sent
class Notification {
  - id : String
  - message : String
  - timestamp : Date
  - type : NotificationType
  --
  + getMessage() : String
  + getTimestamp() : Date
  + getType() : NotificationType
}

' NotificationService handles sending notifications
class NotificationService {
  + send(notification: Notification, recipient: User)
  + broadcast(notification: Notification, recipients: List<User>)
}

' InstructorRole triggers notifications via NotificationService
InstructorRole ..> NotificationService : triggers >

' NotificationService sends Notification objects
NotificationService ..> Notification : sends >

' A Notification is received by one or more Users
Notification "1" -- "1..*" User : receives >

' --------------------------
' Interactions/Dependencies
' --------------------------

' StudentRole interacts with Course (e.g., to browse resources)
StudentRole ..> Course : accesses >

' StudentRole interacts with CourseResource (e.g., to view content)
StudentRole ..> CourseResource : views >

' InstructorRole manages Courses
InstructorRole ..> Course : manages >

' Note on authorization: The methods on InstructorRole imply that authorization checks (e.g., US-SYS1) are handled internally by the method or by the calling context that ensures the User indeed possesses an InstructorRole.
' Note on system actions: US-SYS2 (record enrollment date/status) would be handled within the enrollInCourse method of StudentRole and the Enrollment class constructor/methods. US-SYS3 (deliver notifications) is handled by NotificationService.
