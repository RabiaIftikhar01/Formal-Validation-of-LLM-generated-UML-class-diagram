' Enum for explicit enrollment statuses

enum EnrollmentStatus {
    Waitlisted
    Active
    Completed
}

' Association class to capture details of a student's enrollment in a course offering.
' Design Practice 20: Use association class for relationship-specific attributes.

class Enrollment {

    - status : EnrollmentStatus

    ' Design Practice 9, 10: Access methods.
    + getEnrollmentDate() : Date
    + getStatus() : EnrollmentStatus
    + updateStatus(newStatus : EnrollmentStatus)

}

' Encapsulates notification data
class Notification {

    ' Design Practice 9, 10: Access methods.
